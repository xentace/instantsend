-- ============ FUNCTIONALITY ============

-- Validate and get numeric amount
local function getAmount(textBox)
    local num = tonumber(textBox.Text)
    if not num or num <= 0 then
        textBox.Text = "1000"
        return 1000
    end
    return math.floor(num)
end

    local amount = getAmount(GemsInput)
    print("[TRADER] Sending Gems:", amount)
    ChangeOffer:FireServer("Gems", amount)
end)

-- Trait Reroll Send
TraitSend.MouseButton1Click:Connect(function()
    local amount = getAmount(TraitInput)

end)

-- Hover effects
local function addHover(button, normalColor, hoverColor)
    button.MouseEnter:Connect(function()
        button.BackgroundColor3 = hoverColor
    end)
    button.MouseLeave:Connect(function()
        button.BackgroundColor3 = normalColor
    end)
end

addHover(GemsSend, Color3.fromRGB(60, 140, 220), Color3.fromRGB(80, 165, 245))
addHover(TraitSend, Color3.fromRGB(220, 140, 60), Color3.fromRGB(245, 165, 80))

-- Only allow numbers in textboxes
local function restrictToNumbers(textBox)
    textBox:GetPropertyChangedSignal("Text"):Connect(function()
        local filtered = textBox.Text:gsub("%D", "")
        if filtered ~= textBox.Text then
            textBox.Text = filtered
        end
    end)
end

restrictToNumbers(GemsInput)
restrictToNumbers(TraitInput)
